$(document).ready(function() {
    let selectedTable = null;
    let currentUser = null;

    // Initialize the page
    initPage();

    // Initialize page elements and event listeners
    function initPage() {
        // Clear any existing modal backdrops
        $('body').removeClass('modal-open');
        $('.modal-backdrop').remove();
        
        checkLoginStatus();
        setupEventListeners();
        setDefaultReservationDate();
    }

// Set up event listeners
function setupEventListeners() {
    // Remove any existing click handlers to prevent duplicates
    $(' #bookTableBtn').off('click');
    
    // Handle Book Table button click
    $('#bookTableBtn').on('click', function(e) {
        e.preventDefault();
        e.stopPropagation();
        
        // Hide reservation modal if it's showing
        $('#reservationModal').modal('hide');
        
        // Show notice modal first
        $('#noticeModal').modal('show');
    });

    // Handle "I Understand" button click
    $('#proceedToReservationBtn').off('click').on('click', function() {
        $('#noticeModal').modal('hide');
        
        // Only show reservation modal after notice modal is fully hidden
        $('#noticeModal').on('hidden.bs.modal', function() {
            $('#reservationModal').modal('show');
            // Remove the event listener to prevent multiple bindings
            $('#noticeModal').off('hidden.bs.modal');
        });
    });
    
    // Rest of your existing event listeners...
    $('#reservationForm').on('submit', handleReservationSubmit);
    $('#loginForm').on('submit', handleLoginSubmit);
    $('#registerForm').on('submit', handleRegisterSubmit);
    
    // Date/Time change handlers
    $('#reservationDate, #reservationTime').on('change', loadAvailableTables);
    
    // Modal shown events
    $('#reservationsModal').on('show.bs.modal', loadReservations);
    
    $('#reservationModal').on('shown.bs.modal', function() {
        const date = $('#reservationDate').val();
        const time = $('#reservationTime').val();
        if (date && time) loadAvailableTables();
        
        // Reset form state
        $('#reservationForm')[0].reset();
        $('.table-card').removeClass('selected bg-primary text-white');
        selectedTable = null;
        setDefaultReservationDate();
    });
    
    // Clear selection when reservation modal hides
    $('#reservationModal').on('hidden.bs.modal', function() {
        $('.table-card').removeClass('selected bg-primary text-white');
        selectedTable = null;
    });
    
    // Initialize logout functionality
    $(document).on('click', '#logoutBtn', function(e) {
        e.preventDefault();
        logout();
    });
    
    // Check initial login status
    checkLoginStatus();
}

    // Set default reservation date to today
    function setDefaultReservationDate() {
        const today = new Date().toISOString().split('T')[0];
        $('#reservationDate').val(today);
    }

    // Handle reservation form submission
    function handleReservationSubmit(e) {
        e.preventDefault();
        
        if (!selectedTable) {
            showAlert('Please select a table', 'warning');
            return;
        }
    
        const reservationData = {
            date: $('#reservationDate').val(),
            time: $('#reservationTime').val(),
            table: selectedTable.number,
            requests: $('#specialRequests').val(),
            created_at: new Date().toISOString(),
            status: 'pending',
            name: $('#reservationName').val(),
            phone: $('#reservationPhone').val(),
            email: $('#reservationEmail').val(),
            occasion: $('#reservationOccasion').val()
        };
    
        // Validate required fields
        if (!reservationData.date || !reservationData.time || 
            !reservationData.name || !reservationData.phone || !reservationData.email) {
            showAlert('Please fill all required fields', 'warning');
            return;
        }
    
        const $submitBtn = $(this).find('[type="submit"]');
        $submitBtn.prop('disabled', true).html('<span class="spinner-border spinner-border-sm"></span> Processing...');
    
        $.ajax({
            url: 'php/reserve.php',
            method: 'POST',
            contentType: 'application/json',
            data: JSON.stringify(reservationData),
            dataType: 'json'
        })
        .done(function(response) {
            if (response && response.success) {
                showReservationSuccess(
                    selectedTable.number,
                    reservationData.date,
                    reservationData.time,
                    'pending'
                );
                $('#reservationModal').modal('hide');
                addNewReservation({
                    table: selectedTable.number,
                    date: reservationData.date,
                    time: reservationData.time,
                    status: 'pending'
                });
            } else {
                const errorMsg = response?.message || 'Reservation failed';
                showAlert(errorMsg, 'danger');
            }
        })
        .fail(function(xhr) {
            let errorMsg = 'Error processing reservation';
            
            try {
                const response = xhr.responseJSON;
                if (response && response.message) {
                    errorMsg = response.message;
                } else if (xhr.statusText) {
                    errorMsg += `: ${xhr.statusText}`;
                }
            } catch (e) {
                console.error('Error parsing error response:', e);
            }
            
            console.error("Reservation error:", xhr.responseText);
            showAlert(errorMsg, 'danger');
        })
        .always(function() {
            $submitBtn.prop('disabled', false).text('Book Table');
        });
    }

    // Handle login form submission
    function handleLoginSubmit(e) {
        e.preventDefault();
        
        const $form = $(this);
        const $submitBtn = $form.find('[type="submit"]');
        const originalBtnText = $submitBtn.html();
        
        $submitBtn.prop('disabled', true).html(
            '<span class="spinner-border spinner-border-sm"></span> Logging in...'
        );
        
        const formData = {
            login: $('#loginInput').val().trim(),
            password: $('#loginPassword').val().trim()
        };
        
        $.ajax({
            url: 'php/login.php',
            method: 'POST',
            contentType: 'application/json',
            data: JSON.stringify(formData),
            dataType: 'json'
        })
        .done(function(response) {
            if (response && response.success) {
                localStorage.setItem('kpopCafeUser', JSON.stringify(response.user));
                updateNavForLoggedInUser();
                $('#loginModal').modal('hide');
                $('#welcomeMessage').text(`Welcome back, ${response.user.username}!`);
                showLoginSuccessModal();
                loadReservations();
            } else {
                showAlert(response.message || "Login failed. Please try again.", 'danger');
            }
        })
        .fail(function(xhr) {
            console.error("Login error:", xhr.responseText);
            showAlert("Login failed. Please try again.", 'danger');
        })
        .always(function() {
            $submitBtn.prop('disabled', false).html(originalBtnText);
        });
    }

    // Handle registration form submission
    function handleRegisterSubmit(e) {
        e.preventDefault();
        
        const $submitBtn = $(this).find('[type="submit"]');
        const originalBtnText = $submitBtn.text();
        $submitBtn.prop('disabled', true).html('<span class="spinner-border spinner-border-sm"></span> Creating account...');
    
        if ($('#registerPassword').val() !== $('#registerConfirmPassword').val()) {
            showAlert('Passwords do not match', 'warning');
            $submitBtn.prop('disabled', false).text(originalBtnText);
            return;
        }
    
        const formData = {

    $time = $_GET['time'] ?? '';
    
    if (empty($date) || empty($time)) {
        throw new Exception('Date and time are required');
    }

    // Get all tables
    $stmt = $pdo->query("SELECT * FROM cafe_tables");
    $allTables = $stmt->fetchAll(PDO::FETCH_ASSOC);
    
    // Get reserved tables for this timeslot
    $stmt = $pdo->prepare("SELECT table_number FROM reservations 
                          WHERE reservation_date = ? 
                          AND reservation_time = ?
                          AND status != 'cancelled'");
    $stmt->execute([$date, $time]);
    $reservedTables = $stmt->fetchAll(PDO::FETCH_COLUMN, 0);
    
    // Mark reserved status
    $tables = array_map(function($table) use ($reservedTables) {
        $table['is_reserved'] = in_array($table['table_number'], $reservedTables);
        return $table;
    }, $allTables);
    
    echo json_encode([
        'success' => true,
        'tables' => $tables
    ]);
    
} catch (Exception $e) {
    http_response_code(400);
    echo json_encode([
        'success' => false,

    ]);

        http_response_code(400); // Bad Request
        throw new Exception('Invalid JSON input');
    }

    // Check required fields
    $required = ['table', 'date', 'time', 'status', 'created_at', 
                'name', 'phone', 'email'];
    foreach ($required as $field) {
        if (empty($input[$field])) {
            http_response_code(400); // Bad Request
            throw new Exception("Missing required field: $field");
        }
    }

    // Check user session
    if (!isset($_SESSION['user_id'])) {
        http_response_code(401); // Unauthorized
        throw new Exception('User not logged in');
    }

    // Check table availability
    $stmt = $pdo->prepare("
        SELECT t.id 
        FROM cafe_tables t
        LEFT JOIN reservations r ON t.table_number = r.table_number 
            AND r.reservation_date = ? 
            AND r.reservation_time = ?
            AND r.status != 'cancelled'
        WHERE t.table_number = ? AND r.id IS NULL
    ");
    $stmt->execute([$input['date'], $input['time'], $input['table']]);
    
    if ($stmt->rowCount() === 0) {
        http_response_code(409); // Conflict
        throw new Exception('Table is not available for the selected time');
    }

    // Create reservation
    $stmt = $pdo->prepare("
        INSERT INTO reservations 
        (user_id, table_number, reservation_date, reservation_time, 
         special_requests, created_at, status, 
         customer_name, customer_phone, customer_email, occasion) 
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    ");
    $success = $stmt->execute([
        $_SESSION['user_id'],
        $input['table'],
        $input['date'],
        $input['time'],
        $input['requests'] ?? '',
        $input['created_at'],
        $input['status'],
        $input['name'],
        $input['phone'],
        $input['email'],
        $input['occasion'] ?? 'none'
    ]);

    if (!$success) {
        http_response_code(500); // Internal Server Error
        throw new Exception('Failed to create reservation');
    }

    // Success response
    echo json_encode([
        'success' => true,
        'message' => 'Reservation successful',
        'reservation' => [
            'id' => $pdo->lastInsertId(),
            'table_number' => $input['table'],
            'reservation_date' => $input['date'],
            'reservation_time' => $input['time'],
            'status' => $input['status']
        ]
    ]);

} catch (Exception $e) {
    // Error response
    echo json_encode([
        'success' => false,
        'message' => $e->getMessage(),
        'error_code' => http_response_code()
    ]);
}

<?php
header('Content-Type: application/json');

try {
    // Database connection
    $conn = new mysqli('localhost', 'root', '', 'kpop_cafe');

    if ($conn->connect_error) {
        throw new Exception('Database connection failed: ' . $conn->connect_error);
    }

    // Get JSON input
    $data = json_decode(file_get_contents('php://input'), true);

    if (!isset($data['table_number'], $data['capacity'], $data['location'])) {
        throw new Exception('Invalid input data');
    }

    // Prepare and execute SQL query (exclude id as it should be auto-incremented)
    $stmt = $conn->prepare("INSERT INTO cafe_tables (table_number, capacity, location) VALUES (?, ?, ?)");
    $stmt->bind_param('iis', $data['table_number'], $data['capacity'], $data['location']);

    if ($stmt->execute()) {
        echo json_encode(['success' => true]);
    } else {
        throw new Exception('Failed to save table details: ' . $stmt->error);
    }

    $stmt->close();
    $conn->close();
} catch (Exception $e) {
    echo json_encode(['success' => false, 'message' => $e->getMessage()]);
}

CREATE DATABASE kpop_cafe;

USE kpop_cafe;

-- Users table
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    password VARCHAR(255) NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- Reservations table
CREATE TABLE reservations (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    reservation_date DATE NOT NULL,
    reservation_time TIME NOT NULL,
    table_number INT NOT NULL,
    special_requests TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id)
);

-- Tables available
CREATE TABLE cafe_tables (
    id INT AUTO_INCREMENT PRIMARY KEY,
    table_number INT NOT NULL UNIQUE,
    capacity INT NOT NULL,
    location VARCHAR(50)
);
