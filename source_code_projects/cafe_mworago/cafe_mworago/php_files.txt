get_tables.php
<?php
require_once 'config.php';

header('Content-Type: application/json');

try {
    // Validate input
    $date = $_GET['date'] ?? '';
    $time = $_GET['time'] ?? '';
    
    if (empty($date) || empty($time)) {
        throw new Exception('Date and time are required');
    }

    // Get all tables
    $stmt = $pdo->query("SELECT * FROM cafe_tables");
    $allTables = $stmt->fetchAll(PDO::FETCH_ASSOC);
    
    // Get reserved tables for this timeslot
    $stmt = $pdo->prepare("SELECT table_number FROM reservations 
                          WHERE reservation_date = ? 
                          AND reservation_time = ?
                          AND status != 'cancelled'");
    $stmt->execute([$date, $time]);
    $reservedTables = $stmt->fetchAll(PDO::FETCH_COLUMN, 0);
    
    // Mark reserved status
    $tables = array_map(function($table) use ($reservedTables) {
        $table['is_reserved'] = in_array($table['table_number'], $reservedTables);
        return $table;
    }, $allTables);
    
    echo json_encode([
        'success' => true,
        'tables' => $tables
    ]);
    
} catch (Exception $e) {
    http_response_code(400);
    echo json_encode([
        'success' => false,
        'message' => $e->getMessage()
    ]);
}
?>

    // Get notifications
    $stmt = $pdo->prepare("SELECT * FROM notifications WHERE user_id = ? ORDER BY created_at DESC LIMIT 5");
    $stmt->execute([$userId]);
    $notifications = $stmt->fetchAll(PDO::FETCH_ASSOC);
    
    // Get recent activities
    $stmt = $pdo->prepare("SELECT * FROM activity_log WHERE user_id = ? ORDER BY created_at DESC LIMIT 5");
    $stmt->execute([$userId]);
    $activities = $stmt->fetchAll(PDO::FETCH_ASSOC);
    
    // Get upcoming reservations
    $stmt = $pdo->prepare("SELECT * FROM reservations 
                          WHERE user_id = ? AND status = 'confirmed' 
                          AND CONCAT(reservation_date, ' ', reservation_time) >= NOW()
                          ORDER BY reservation_date, reservation_time LIMIT 3");
    $stmt->execute([$userId]);
    $reservations = $stmt->fetchAll(PDO::FETCH_ASSOC);
    
    echo json_encode([
        'success' => true,
        'notifications' => $notifications,
        'activities' => $activities,
        'reservations' => $reservations
    ]);
} catch (PDOException $e) {
    http_response_code(500);
    echo json_encode(['success' => false, 'message' => 'Database error: ' . $e->getMessage()]);
}

login.php
<?php
require_once 'config.php';

header('Content-Type: application/json');

// Get input
$data = json_decode(file_get_contents('php://input'), true);
$login = $data['login'] ?? '';
$password = $data['password'] ?? '';

// Validate
if (empty($login) || empty($password)) {
    http_response_code(400);
    echo json_encode(['success' => false, 'message' => 'All fields are required']);
    exit;
}

try {
    // Database query
    $stmt = $pdo->prepare("SELECT * FROM users WHERE username = ? OR email = ?");
    $stmt->execute([$login, $login]);
    $user = $stmt->fetch();
    
    if ($user && password_verify($password, $user['password'])) {
        // Successful login
        session_regenerate_id(true);
        $_SESSION['user_id'] = $user['id'];
        $_SESSION['username'] = $user['username'];
        
        unset($user['password']);
        echo json_encode([
            'success' => true,
            'user' => $user,
            'message' => 'Login successful'
        ]);
    } else {
        http_response_code(401);
        echo json_encode(['success' => false, 'message' => 'Invalid credentials']);
    }
} catch (PDOException $e) {
    http_response_code(500);
    echo json_encode(['success' => false, 'message' => 'Database error']);
}

<?php
require_once 'config.php';

// Set header first
header('Content-Type: application/json');

// Get input data
$data = json_decode(file_get_contents('php://input'), true);

// Validate input
if (empty($data['username']) || empty($data['email']) || empty($data['password'])) {
    http_response_code(400);
    echo json_encode([
        'success' => false,
        'message' => 'All fields are required',
        'received_data' => $data // For debugging
    ]);
    exit;
}

try {
    // Check if user exists
    $stmt = $pdo->prepare("SELECT id FROM users WHERE username = ? OR email = ?");
    $stmt->execute([$data['username'], $data['email']]);
    
    if ($stmt->rowCount() > 0) {
        http_response_code(400);
        echo json_encode([
            'success' => false,
            'message' => 'Username or email already exists'
        ]);
        exit;
    }
    
    // Hash password
    $hashedPassword = password_hash($data['password'], PASSWORD_DEFAULT);
    
    // Insert user
    $stmt = $pdo->prepare("INSERT INTO users (username, email, password) VALUES (?, ?, ?)");
    $stmt->execute([$data['username'], $data['email'], $hashedPassword]);
    
    // Return success with user data (excluding password)
    echo json_encode([
        'success' => true,
        'message' => 'Registration successful',
        'user' => [
            'id' => $pdo->lastInsertId(),
            'username' => $data['username'],
            'email' => $data['email']
        ]
    ]);
    exit;

} catch (PDOException $e) {
    http_response_code(500);
    echo json_encode([
        'success' => false,
        'message' => 'Database error',
        'error' => $e->getMessage()
    ]);
    exit;
}
?>
reserve.php
<?php
ini_set('display_errors', 1);
ini_set('display_startup_errors', 1);
error_reporting(E_ALL);

require_once 'config.php';
header("Access-Control-Allow-Origin: *");
header("Access-Control-Allow-Methods: POST");
header("Access-Control-Allow-Headers: Content-Type");
header('Content-Type: application/json');
try {
    $pdo->query("SELECT 1");
} catch (PDOException $e) {
    die(json_encode(['success' => false, 'message' => 'Database connection failed']));
}
try {
    // Verify request method
    if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
        http_response_code(405); // Method Not Allowed
        throw new Exception('Invalid request method');
    }

    // Get and validate input
    $input = json_decode(file_get_contents('php://input'), true);
    if (json_last_error() !== JSON_ERROR_NONE) {
        http_response_code(400); // Bad Request
        throw new Exception('Invalid JSON input');
    }

    // Check required fields
    $required = ['table', 'date', 'time', 'status', 'created_at', 
                'name', 'phone', 'email'];
    foreach ($required as $field) {
        if (empty($input[$field])) {
            http_response_code(400); // Bad Request
            throw new Exception("Missing required field: $field");
        }
    }

    // Check user session
    if (!isset($_SESSION['user_id'])) {
        http_response_code(401); // Unauthorized
        throw new Exception('User not logged in');
    }

    // Check table availability
    $stmt = $pdo->prepare("
        SELECT t.id 
        FROM cafe_tables t
        LEFT JOIN reservations r ON t.table_number = r.table_number 
            AND r.reservation_date = ? 
            AND r.reservation_time = ?
            AND r.status != 'cancelled'
        WHERE t.table_number = ? AND r.id IS NULL
    ");
    $stmt->execute([$input['date'], $input['time'], $input['table']]);
    
    if ($stmt->rowCount() === 0) {
        http_response_code(409); // Conflict
        throw new Exception('Table is not available for the selected time');
    }

    // Create reservation
    $stmt = $pdo->prepare("
        INSERT INTO reservations 
        (user_id, table_number, reservation_date, reservation_time, 
         special_requests, created_at, status, 
         customer_name, customer_phone, customer_email, occasion) 
        VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    ");
    $success = $stmt->execute([
        $_SESSION['user_id'],
        $input['table'],
        $input['date'],
        $input['time'],
        $input['requests'] ?? '',
        $input['created_at'],
        $input['status'],
        $input['name'],
        $input['phone'],
        $input['email'],
        $input['occasion'] ?? 'none'
    ]);

    if (!$success) {
        http_response_code(500); // Internal Server Error
        throw new Exception('Failed to create reservation');
    }

    // Success response
    echo json_encode([
        'success' => true,
        'message' => 'Reservation successful',
        'reservation' => [
            'id' => $pdo->lastInsertId(),
            'table_number' => $input['table'],
            'reservation_date' => $input['date'],
            'reservation_time' => $input['time'],
            'status' => $input['status']
        ]
    ]);

} catch (Exception $e) {
    // Error response
    echo json_encode([
        'success' => false,
        'message' => $e->getMessage(),
        'error_code' => http_response_code()
    ]);
}
?>
