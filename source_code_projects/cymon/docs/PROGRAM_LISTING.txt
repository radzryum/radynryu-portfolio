A.1  Architectural Pattern
--------------------------------------------------------------------------------
Decoupled three-tier architecture: a single-page application for presentation,
a stateless JSON API for application logic, and a managed PostgreSQL instance
whose access is governed entirely by Row-Level Security.

    +---------------------------+         +-----------------------------+
    |  PRESENTATION TIER        |         |  APPLICATION TIER           |
    |  React 19 SPA (Vite 8)    |  HTTPS  |  Express 5 (Node.js)       |
    |  :5173  Tailwind CSS v4   | ------> |  :4000  /api/*             |
    |  30 UI primitives         |   JSON  |  routers > middleware >    |
    +---------------------------+         |  services (lib/)            |
             ^                            +-------------+---------------+
             |  Bearer access token                      | service-role key
             +-----------------------------+  +-----------+--------------+
                                         |  DATA TIER               |
                                         |  Supabase PostgreSQL     |
                                         |  + Auth (GoTrue)         |
                                         |  + Row-Level Security    |
                                         +--------------------------+

In development the Vite dev server proxies /api/* to the backend, so the browser
only issues same-origin requests (frontend/vite.config.js):

    server: { proxy: { '/api': { target: 'http://localhost:4000',
                                 changeOrigin: true } } }

A.2  Technology Stack
--------------------------------------------------------------------------------
  Client UI   React 19.2, React Router 7, Vite 8, Tailwind CSS v4, lucide-react
  Client Auth @supabase/supabase-js (anon key, persistSession: true)
  API Server  Node.js, Express 5, cors, dotenv
  PDF Engine  pdf-lib (PDFDocument, StandardFonts, PNG signature embedding)
  Services    Resend (transactional e-mail), Vercel Cron (reminders)
  Data        Supabase PostgreSQL 15, 73 forward-only SQL migrations
  Test / CI   Playwright (9 e2e specs), GitHub Actions, Vercel deploy

A.3  Roles and Portals
--------------------------------------------------------------------------------
The enum `user_role` carries six values. A staff member may hold more than one
role at a time (profiles.extra_roles), so the interface shows an acting-role
switcher and re-scopes navigation without re-authenticating.

  client                  /client            Caregiver: journal, waivers,
                                             releases, appointments
  psychologist            /psychologist      Approval and clinical sign-off
  psychometrician         /psychometrician   Administration, report drafting
  speech_therapist        /psychometrician   Shares the psychometrician portal
  occupational_therapist  /psychometrician   Shares the psychometrician portal
  admin                   /admin             Clinic administration, records

    async function requireAuth(req, res, next) {
      const header = req.headers.authorization || '';
      const token = header.startsWith('Bearer ') ? header.slice(7) : null;
      if (!token) return res.status(401).json({ error: 'Missing bearer token' });
      const { data, error } = await supabase.auth.getUser(token);
      if (error || !data.user) {
        return res.status(401).json({ error: 'Invalid or expired token' });
      }
      // Progressive column fallback so the API survives a partially applied
      // migration set (extra_roles / onboarding columns may not exist yet).
      const { data: profile } = await supabase.from('profiles')
        .select('id, role, extra_roles, clinic_id, deleted_at, display_name')
        .eq('id', data.user.id).single();

    async function createNotification({ clinicId, recipientId, type = 'system',
                                        title, body = null, link = null, email = null }) {
      if (!recipientId || !clinicId || !title) return;
      const { error } = await supabase.from('notifications').insert({
        clinic_id: clinicId, recipient_id: recipientId, type, title, body, link });
      if (error) console.error('[notify] failed to create notification:', error.message);
      if (!email || error) return;
      const { data: recipient } = await supabase.from('profiles')
        .select('email, display_name').eq('id', recipientId).maybeSingle();
      if (!recipient?.email) return;
      sendMail({ to: recipient.email, subject: email.subject || title,
        templateKey: email.templateKey || 'notification',
        templateData: { recipient_name: recipient.display_name || 'there',
          notification_title: title, notification_body: body || '',
          notification_link: link || '', ...(email.templateData || {}) },
      }).catch((e) => console.error('[email] failed to send notification:', e.message));

        entity_id: entityId || null, summary: String(summary).trim(),
        severity: ALLOWED_SEVERITIES.has(severity) ? severity : 'info',

    const allowedOrigins = (process.env.CORS_ORIGIN ||
      'http://localhost:5173,http://localhost:5174').split(',')
      .map((o) => o.trim()).filter(Boolean);
    app.use(cors({
      origin: (origin, callback) => {
        if (!origin) return callback(null, true);
        if (allowedOrigins.includes('*') || allowedOrigins.includes(origin)) {
          return callback(null, true);
        }
        return callback(new Error(`Origin ${origin} not allowed by CORS`));
      },
      credentials: true,
    }));
    app.use(express.json({ limit: '10mb' }));
    app.get('/api/health', (req, res) => {
      res.json({ status: 'ok', uptime: process.uptime(), allowedOrigins });
    });
    // Central error middleware — never return 200 with an { error } payload.
    app.use((err, req, res, _next) => {
      console.error(err);
      const status = err.status || 500;
      res.status(status).json({ error: err.message || 'Internal Server Error' });
    });

    async function request(path, { method = 'GET', body, auth = false,
                                   accessToken: suppliedToken = null } = {}) {
      const headers = { 'Content-Type': 'application/json' }
      if (auth) {
        const accessToken = suppliedToken || getAccessToken()
        if (!accessToken) {
          throw new Error('Your session expired or is missing. Please sign in again.')
        }
        headers.Authorization = `Bearer ${accessToken}`
      }
      const res = await fetch(`${BASE}${path}`, {
        method, headers, body: body ? JSON.stringify(body) : undefined })
      const text = await res.text()
      const data = text ? JSON.parse(text) : null
      if (!res.ok) {
        const message = data?.error || `Request failed (${res.status})`
        const err = new Error(message); err.status = res.status; throw err
      }
      return data
    }
    async function download(path, { auth = false } = {}) {
      const headers = {}
      if (auth) {
        const accessToken = getAccessToken()
        if (!accessToken) {
          throw new Error('Your session expired or is missing. Please sign in again.')
        }
        headers.Authorization = `Bearer ${accessToken}`
      }
      const res = await fetch(`${BASE}${path}`, { headers })
      if (!res.ok) {
        const text = await res.text()
        let data = null
        try { data = text ? JSON.parse(text) : null } catch { /* may not be JSON */ }
        const err = new Error(data?.error || `Download failed (${res.status})`)
        err.status = res.status; throw err
      }
      return res.blob()
    }
